-- 오방도감 Supabase 첫 설정 (초안 10/4, 은주가 SQL Editor에 붙여 넣어 실행)
-- 회원 프로필 · 보관함 · 구매 기록. 모든 표에 행 단위 접근 제한(RLS): 회원은 자기 것만.
-- 구매 기록은 화면이 쓰지 못하고 서버 함수(service_role)만 쓴다. 금액 · 상품은 서버 상품표가 정한다.

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  nickname text,
  avatar_url text,
  provider text,
  marketing_ok boolean not null default false,   -- 마케팅 수신 동의(기본 꺼짐)
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.archive (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  menu text not null,              -- 메뉴 키(예: myeongri, sinnyeon)
  title text,                      -- 보관함 카드 제목
  sub text,                        -- 한 줄 요약
  url text,                        -- 다시 열기 주소
  form jsonb,                      -- 입력값(생년월일 · 이름 · 물음 · 사연 등, 결과에 필요한 만큼만)
  body jsonb,                      -- 결과 본문(유료 AI 원고 포함)
  paid boolean not null default false,
  created_at timestamptz not null default now()
);
create index if not exists archive_user_idx on public.archive(user_id, created_at desc);

create table if not exists public.purchases (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete restrict,
  product text not null,           -- 상품표 키
  amount integer not null,         -- 서버가 정한 금액(원)
  order_id text unique not null,   -- 주문번호(중복 처리 방지)
  receipt_id text,                 -- 부트페이 영수증 id
  status text not null default 'paid' check (status in ('paid','refunded','partial_refund')),
  archive_id uuid references public.archive(id) on delete set null,
  created_at timestamptz not null default now(),
  refunded_at timestamptz
);
create index if not exists purchases_user_idx on public.purchases(user_id, created_at desc);

alter table public.profiles  enable row level security;
alter table public.archive   enable row level security;
alter table public.purchases enable row level security;

-- 프로필: 자기 것만 보고 고침
create policy "profiles self select" on public.profiles for select using (auth.uid() = id);
create policy "profiles self update" on public.profiles for update using (auth.uid() = id);
create policy "profiles self insert" on public.profiles for insert with check (auth.uid() = id);

-- 보관함: 자기 것만 보고, 무료 결과만 직접 넣고, 직접 지울 수 있음(유료 결과는 서버가 넣음)
create policy "archive self select" on public.archive for select using (auth.uid() = user_id);
create policy "archive self insert free" on public.archive for insert with check (auth.uid() = user_id and paid = false);
create policy "archive self delete" on public.archive for delete using (auth.uid() = user_id);

-- 구매 기록: 자기 것 보기만(쓰기는 서버 함수 service_role만, 정책 없음 = 화면에서 불가)
create policy "purchases self select" on public.purchases for select using (auth.uid() = user_id);

-- 가입하면 프로필 한 줄 자동 생성
create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, nickname, avatar_url, provider)
  values (new.id,
          coalesce(new.raw_user_meta_data->>'name', new.raw_user_meta_data->>'nickname', new.raw_user_meta_data->>'full_name'),
          coalesce(new.raw_user_meta_data->>'avatar_url', new.raw_user_meta_data->>'picture'),
          new.raw_app_meta_data->>'provider')
  on conflict (id) do nothing;
  return new;
end $$;
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_user();
